local function Create(Class_Name: string, Properties: { [string]: any }): Instance
	local Inst = Instance.new(Class_Name)
	for K, V in pairs(Properties) do
		(Inst :: any)[K] = V
	end
	return Inst
end

local Preloaded_Icons: { [string]: boolean } = {}
local Pending_Preload: { string } = {}
local Preload_Scheduled = false

local function Queue_Preload(Asset_Id: string)
	if Preloaded_Icons[Asset_Id] then return end
	Preloaded_Icons[Asset_Id] = true
	table.insert(Pending_Preload, Asset_Id)
	if not Preload_Scheduled then
		Preload_Scheduled = true
		task.defer(function()
			Preload_Scheduled = false
			local Batch = Pending_Preload
			Pending_Preload = {}
			task.spawn(function()
				pcall(function()
					ContentProvider:PreloadAsync(Batch)
				end)
			end)
		end)
	end
end

local function Get_Icon(Name: string?): string
	if not Name or Name == "" then return "" end
	local Id = LucideIcons[Name] or LucideIcons["lucide-" .. Name]
	if not Id then return "" end
	Queue_Preload(Id)
	return Id
end

local function Tween(Inst: Instance, Properties: { [string]: any }, Duration: number?, Style: Enum.EasingStyle?, Direction: Enum.EasingDirection?): Tween
	Duration = Duration or 0.25
	Style = Style or Enum.EasingStyle.Quint
	Direction = Direction or Enum.EasingDirection.Out
	local Tween_Info = TweenInfo.new(Duration :: number, Style :: Enum.EasingStyle, Direction :: Enum.EasingDirection)
	local Tween_Obj = TweenService:Create(Inst, Tween_Info, Properties)
	Tween_Obj:Play()
	return Tween_Obj
end

local function Make_Draggable(Top_Bar: Frame, Main: Frame)
	local Is_Dragging = false
	local Drag_Input: InputObject? = nil
	local Drag_Start: Vector3? = nil
	local Start_Pos: UDim2? = nil

	Top_Bar.InputBegan:Connect(function(Input: InputObject)
		if Input.UserInputType == Enum.UserInputType.MouseButton1 or Input.UserInputType == Enum.UserInputType.Touch then
			Is_Dragging = true
			Drag_Start = Input.Position
			Start_Pos = Main.Position
			Input.Changed:Connect(function()
				if Input.UserInputState == Enum.UserInputState.End then
					Is_Dragging = false
				end
			end)
		end
	end)

	Top_Bar.InputChanged:Connect(function(Input: InputObject)
		if Input.UserInputType == Enum.UserInputType.MouseMovement or Input.UserInputType == Enum.UserInputType.Touch then
			Drag_Input = Input
		end
	end)

	UserInputService.InputChanged:Connect(function(Input: InputObject)
		if Input == Drag_Input and Is_Dragging and Drag_Start and Start_Pos then
			local Delta = Input.Position - Drag_Start
			Main.Position = UDim2.new(Start_Pos.X.Scale, Start_Pos.X.Offset + Delta.X, Start_Pos.Y.Scale, Start_Pos.Y.Offset + Delta.Y)
		end
	end)
end

local function Color_To_Hex(C: Color3): string
	return string.format("%02X%02X%02X", math.floor(C.R * 255 + 0.5), math.floor(C.G * 255 + 0.5), math.floor(C.B * 255 + 0.5))
end

local function Hex_To_Color(Hex: string): Color3?
	local Clean_Hex = Hex:gsub("#", ""):gsub("%s", "")
	if #Clean_Hex ~= 6 then return nil end
	local R = tonumber(Clean_Hex:sub(1, 2), 16)
	local G = tonumber(Clean_Hex:sub(3, 4), 16)
	local B = tonumber(Clean_Hex:sub(5, 6), 16)
	if not (R and G and B) then return nil end
	return Color3.fromRGB(R, G, B)
end

				local Percent = math.clamp((Slider.Value - Min) / (Max - Min), 0, 1)
				Tween(Fill, { Size = UDim2.new(Percent, 0, 1, 0) }, 0.1)
				Tween(Knob, { Position = UDim2.new(Percent, -6, 0.5, -6) }, 0.1)
				Value_Lbl.Text = tostring(Slider.Value)
			end
			function Slider:OnChanged(Cb: Callback) table.insert(self.Callbacks, Cb) end
			function Slider:SetValue(Val: number)
				self.Value = math.clamp(Val, Min, Max)
				Update_Visuals()
				for _, Cb in pairs(self.Callbacks) do Cb(self.Value) end
			end

			local Dragging = false
			Btn.InputBegan:Connect(function(Input: InputObject)
				if Input.UserInputType == Enum.UserInputType.MouseButton1 or Input.UserInputType == Enum.UserInputType.Touch then Dragging = true end

					Slider:SetValue(Val)
				end
			end)
			Update_Visuals()
			NullUL.Options[Name] = Slider
			return Slider
		end

			local function Update_Visuals()
				if Multi then
					local Selected = {}
					for K, V in pairs(Dropdown.Value) do if V then table.insert(Selected, K) end end
					Selected_Lbl.Text = #Selected > 0 and table.concat(Selected, ", ") or "None"
				else
					Selected_Lbl.Text = tostring(Dropdown.Value)
				end
			end

			local function Build_List()
				for _, Child in pairs(List_Container:GetChildren()) do if Child:IsA("TextButton") then Child:Destroy() end end
				for _, Val in ipairs(Opts.Values or {}) do
					local Item_Btn = Create("TextButton", { Parent = List_Container, Size = UDim2.new(1, 0, 0, 30), BackgroundColor3 = Theme.Background, Text = "", AutoButtonColor = false }) :: TextButton
					Create("UICorner", { Parent = Item_Btn, CornerRadius = UDim.new(0, 4) })

						local Is_Selected = Multi and Dropdown.Value[Val] or (not Multi and Dropdown.Value == Val)
						Check.Visible = Is_Selected
						Item_Txt.TextColor3 = Is_Selected and Theme.Accent or Theme.SubText
					end
					Item_Btn.MouseEnter:Connect(function() Tween(Item_Btn, { BackgroundColor3 = Theme.ElementHover }, 0.12) end)
					Item_Btn.MouseLeave:Connect(function() Tween(Item_Btn, { BackgroundColor3 = Theme.Background }, 0.12) end)
					Item_Btn.MouseButton1Click:Connect(function()
						if Multi then Dropdown.Value[Val] = not Dropdown.Value[Val]
						else Dropdown.Value = Val; Expanded = false; Tween(Dd_Frame, { Size = UDim2.new(1, 0, 0, 40) }, 0.22); Tween(Icon, { Rotation = 0 }, 0.22) end
						Update_Item(); Update_Visuals(); for _, Cb in pairs(Dropdown.Callbacks) do Cb(Dropdown.Value) end
					end)
					Update_Item()
				end
			end

			Btn.MouseButton1Click:Connect(function()
				Expanded = not Expanded
				local Target_Size = Expanded and UDim2.new(1, 0, 0, 50 + List_Layout.AbsoluteContentSize.Y) or UDim2.new(1, 0, 0, 40)
				Tween(Dd_Frame, { Size = Target_Size }, 0.22)
				Tween(Icon, { Rotation = Expanded and 180 or 0 }, 0.22)
			end)

			function Dropdown:OnChanged(Cb: Callback) table.insert(self.Callbacks, Cb) end
			function Dropdown:SetValue(Val: any)
				if Multi and type(Val) == "table" then for K, V in pairs(Val) do self.Value[K] = V end else self.Value = Val end
				Update_Visuals(); Build_List(); for _, Cb in pairs(self.Callbacks) do Cb(self.Value) end
			end

			if Multi and type(Dropdown.Value) ~= "table" then
				Dropdown.Value = {}
				if type(Opts.Default) == "table" then for _, V in ipairs(Opts.Default) do Dropdown.Value[V] = true end end
			end

			Build_List(); Update_Visuals()
			NullUL.Options[Name] = Dropdown
			return Dropdown
		end

			Sv_Cursor_Hit.InputBegan:Connect(function(Input: InputObject)
				if Input.UserInputType == Enum.UserInputType.MouseButton1 or Input.UserInputType == Enum.UserInputType.Touch then
					Dragging_Sv = true; Update_Sv(Input.Position)
				end
			end)
			Hue_Hit.InputBegan:Connect(function(Input: InputObject)
				if Input.UserInputType == Enum.UserInputType.MouseButton1 or Input.UserInputType == Enum.UserInputType.Touch then
					Dragging_Hue = true; Update_Hue(Input.Position)
				end
			end)
			Alpha_Hit.InputBegan:Connect(function(Input: InputObject)
				if Input.UserInputType == Enum.UserInputType.MouseButton1 or Input.UserInputType == Enum.UserInputType.Touch then
					Dragging_Alpha = true; Update_Alpha(Input.Position)
				end
			end)
			UserInputService.InputEnded:Connect(function(Input: InputObject)
				if Input.UserInputType == Enum.UserInputType.MouseButton1 or Input.UserInputType == Enum.UserInputType.Touch then
					Dragging_Sv, Dragging_Hue, Dragging_Alpha = false, false, false
				end
			end)
			UserInputService.InputChanged:Connect(function(Input: InputObject)
				if Input.UserInputType == Enum.UserInputType.MouseMovement or Input.UserInputType == Enum.UserInputType.Touch then
					if Dragging_Sv then Update_Sv(Input.Position)
					elseif Dragging_Hue then Update_Hue(Input.Position)
					elseif Dragging_Alpha then Update_Alpha(Input.Position) end
				end
			end)

			Hex_Box.FocusLost:Connect(function()
				local Parsed = Hex_To_Color(Hex_Box.Text)
				if Parsed then
					H, S, V = Color3.toHSV(Parsed)
					Refresh(true)
				else
					Hex_Box.Text = Color_To_Hex(Colorpicker.Value)
				end
			end)

			Header_Btn.MouseButton1Click:Connect(function()
				Expanded = not Expanded
				local Target_Size = Expanded and UDim2.new(1, 0, 0, 216) or UDim2.new(1, 0, 0, 40)
				Tween(Cp_Frame, { Size = Target_Size }, 0.25, Enum.EasingStyle.Quint)
			end)

			function Colorpicker:OnChanged(Cb: Callback) table.insert(self.Callbacks, Cb) end
			function Colorpicker:SetValueRGB(Color: Color3)
				H, S, V = Color3.toHSV(Color)
				Refresh()
			end
			function Colorpicker:SetValueTransparency(T: number)
				Alpha = 1 - T
				Refresh()
			end

			Sv_Cursor.Position = UDim2.new(S, 0, 1 - V, 0)
			Hue_Cursor.Position = UDim2.new(0, -2, H, 0)
			Alpha_Cursor.Position = UDim2.new(0, -2, 1 - Alpha, 0)

			NullUL.Options[Name] = Colorpicker
			return Colorpicker
		end

		function Tab:AddKeybind(Name: string, Opts: ElementConfig?)
			Opts = Opts or {}
			local Keybind = { Value = Opts.Default or "None", Mode = Opts.Mode or "Toggle", Callbacks = {}, ClickCallbacks = {} }
			local State, Binding = false, false
			local Kb_Frame = Create("Frame", { Parent = Wrapper, Size = UDim2.new(1, 0, 0, 40), BackgroundColor3 = Theme.Element })
			Create("UICorner", { Parent = Kb_Frame, CornerRadius = UDim.new(0, 6) })
			Create("UIStroke", { Parent = Kb_Frame, Color = Theme.Border, Thickness = 1 })
			Create("TextLabel", { Parent = Kb_Frame, Size = UDim2.new(1, -100, 1, 0), Position = UDim2.new(0, 10, 0, 0), BackgroundTransparency = 1, Text = Opts.Title or "Keybind", TextColor3 = Theme.Text, TextSize = 14, Font = Enum.Font.GothamMedium, TextXAlignment = Enum.TextXAlignment.Left })
			local Bind_Btn = Create("TextButton", { Parent = Kb_Frame, Size = UDim2.new(0, 80, 0, 24), Position = UDim2.new(1, -90, 0.5, -12), BackgroundColor3 = Theme.Background, Text = tostring(Keybind.Value), TextColor3 = Theme.SubText, TextSize = 12, Font = Enum.Font.Gotham }) :: TextButton
			Create("UICorner", { Parent = Bind_Btn, CornerRadius = UDim.new(0, 4) })

			Bind_Btn.MouseButton1Click:Connect(function() Binding = true; Bind_Btn.Text = "..."; Bind_Btn.TextColor3 = Theme.Accent end)
			UserInputService.InputBegan:Connect(function(Input: InputObject, Game_Processed: boolean)
				if Binding then
					if Input.UserInputType == Enum.UserInputType.Keyboard then Keybind.Value = Input.KeyCode.Name
					elseif Input.UserInputType == Enum.UserInputType.MouseButton1 then Keybind.Value = "MB1"
					elseif Input.UserInputType == Enum.UserInputType.MouseButton2 then Keybind.Value = "MB2" end
					Binding = false; Bind_Btn.Text = Keybind.Value; Bind_Btn.TextColor3 = Theme.SubText
					for _, Cb in pairs(Keybind.Callbacks) do Cb(Keybind.Value) end
					if Opts.ChangedCallback then Opts.ChangedCallback(Input.KeyCode) end
				elseif not Game_Processed then
					local Match = false
					if Input.UserInputType == Enum.UserInputType.Keyboard and Input.KeyCode.Name == Keybind.Value then Match = true end
					if Input.UserInputType == Enum.UserInputType.MouseButton1 and Keybind.Value == "MB1" then Match = true end
					if Input.UserInputType == Enum.UserInputType.MouseButton2 and Keybind.Value == "MB2" then Match = true end
					if Match then
						if Keybind.Mode == "Toggle" then State = not State; for _, Cb in pairs(Keybind.ClickCallbacks) do Cb(State) end; if Opts.Callback then Opts.Callback(State) end
						elseif Keybind.Mode == "Hold" then State = true; if Opts.Callback then Opts.Callback(State) end end
					end
				end
			end)
			UserInputService.InputEnded:Connect(function(Input: InputObject, Game_Processed: boolean)
				if not Game_Processed and Keybind.Mode == "Hold" then
					local Match = (Input.UserInputType == Enum.UserInputType.Keyboard and Input.KeyCode.Name == Keybind.Value) or (Input.UserInputType == Enum.UserInputType.MouseButton1 and Keybind.Value == "MB1") or (Input.UserInputType == Enum.UserInputType.MouseButton2 and Keybind.Value == "MB2")
					if Match then State = false; if Opts.Callback then Opts.Callback(State) end end
				end
			end)
			function Keybind:OnChanged(Cb: Callback) table.insert(self.Callbacks, Cb) end
			function Keybind:OnClick(Cb: Callback) table.insert(self.ClickCallbacks, Cb) end
			function Keybind:GetState() return State end
			function Keybind:SetValue(Key: string, Mode: string?) self.Value = Key; if Mode then self.Mode = Mode end; Bind_Btn.Text = Key end
			NullUL.Options[Name] = Keybind
			return Keybind
		end

		function Tab:AddInput(Name: string, Opts: ElementConfig?)
			Opts = Opts or {}
			local Input_Obj = { Value = Opts.Default or "", Callbacks = {} }
			local Inp_Frame = Create("Frame", { Parent = Wrapper, Size = UDim2.new(1, 0, 0, 40), BackgroundColor3 = Theme.Element })

			Create("UICorner", { Parent = Text_Box, CornerRadius = UDim.new(0, 4) })

			Text_Box.FocusLost:Connect(function(Enter_Pressed: boolean)
				if Opts.Finished and not Enter_Pressed then return end

				Input_Obj.Value = Text_Box.Text; for _, Cb in pairs(Input_Obj.Callbacks) do Cb(Input_Obj.Value) end; if Opts.Callback then Opts.Callback(Input_Obj.Value) end
			end)
			function Input_Obj:OnChanged(Cb: Callback) table.insert(self.Callbacks, Cb) end
			NullUL.Options[Name] = Input_Obj
			return Input_Obj
		end

		return Tab
	end

	function Window:SelectTab(Tab_Or_Index: any)
		local Target_Tab = type(Tab_Or_Index) == "number" and Tabs[Tab_Or_Index] or Tab_Or_Index
		if not Target_Tab or Target_Tab == Current_Tab then return end
		local Previous_Tab = Current_Tab
		Current_Tab = Target_Tab

		if Previous_Tab then
			Tween(Previous_Tab.Button, { BackgroundTransparency = 1 }, 0.2)
			Tween(Previous_Tab.IconImg, { ImageColor3 = Theme.SubText }, 0.2)
			Tween(Previous_Tab.TitleLbl, { TextColor3 = Theme.SubText }, 0.2)
		end
		Target_Tab.Button.BackgroundColor3 = Theme.Element
		Tween(Target_Tab.Button, { BackgroundTransparency = 0 }, 0.2)
		Tween(Target_Tab.IconImg, { ImageColor3 = Theme.Accent }, 0.2)
		Tween(Target_Tab.TitleLbl, { TextColor3 = Theme.Accent }, 0.2)

		if Previous_Tab then
			local Old_Wrap = Previous_Tab.Wrapper
			Tween(Old_Wrap, { GroupTransparency = 1, Position = UDim2.new(0, -16, 0, 0) }, 0.16)
			task.delay(0.16, function()
				if Current_Tab ~= Previous_Tab then
					Previous_Tab.Page.Visible = false
					Old_Wrap.Position = UDim2.new(0, 0, 0, 0)
					Old_Wrap.GroupTransparency = 0
				end
			end)
		end

		local New_Wrap = Target_Tab.Wrapper
		Target_Tab.Page.Visible = true
		New_Wrap.GroupTransparency = 1
		New_Wrap.Position = UDim2.new(0, 16, 0, 0)
		Tween(New_Wrap, { GroupTransparency = 0, Position = UDim2.new(0, 0, 0, 0) }, 0.24, Enum.EasingStyle.Quint)
	end

					Tween(Overlay, { BackgroundTransparency = 1 }, 0.15)
					Tween(Dialog_Frame, { Position = UDim2.fromScale(0.5, 0.56) }, 0.15)
					task.delay(0.15, function() Overlay:Destroy() end)
				end)
			end
		end
		Tween(Overlay, { BackgroundTransparency = 0.5 }, 0.2)
		Dialog_Frame.Position = UDim2.fromScale(0.5, 0.56)
		Tween(Dialog_Frame, { Position = UDim2.fromScale(0.5, 0.5) }, 0.22, Enum.EasingStyle.Back)
	end
